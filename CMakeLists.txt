cmake_minimum_required(VERSION 3.20)
project(OnlineStoreOrders LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" config_upper)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config_upper} "${CMAKE_BINARY_DIR}/bin")
endforeach()

add_library(store_core backend/src/Store.cpp)
target_include_directories(store_core PUBLIC backend/include)
if(MSVC)
    target_compile_options(store_core PRIVATE /W4)
else()
    target_compile_options(store_core PRIVATE -Wall -Wextra)
endif()

add_executable(store backend/src/main.cpp)
target_link_libraries(store PRIVATE store_core)

find_package(Threads REQUIRED)
add_executable(server backend/src/server.cpp)
target_include_directories(server PRIVATE backend/third_party)
target_link_libraries(server PRIVATE store_core Threads::Threads)
if(WIN32)
    target_compile_definitions(server PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(server PRIVATE ws2_32)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(store_tests backend/tests/tests.cpp)
    target_link_libraries(store_tests PRIVATE store_core)
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/test-data")
    configure_file(data/sample/orders.csv "${CMAKE_BINARY_DIR}/test-data/orders.csv" COPYONLY)
    configure_file(data/sample/delivery_routes.csv "${CMAKE_BINARY_DIR}/test-data/delivery_routes.csv" COPYONLY)
    configure_file(data/test/test_routes_intermediate.csv "${CMAKE_BINARY_DIR}/test-data/test_routes_intermediate.csv" COPYONLY)
    add_test(NAME store_regression COMMAND store_tests)
    set_tests_properties(store_regression PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/test-data")
endif()

# Seed only absent files; configure/build never resets existing runtime data.
file(MAKE_DIRECTORY "${CMAKE_SOURCE_DIR}/data/runtime")
foreach(name orders.csv delivery_routes.csv)
    if(NOT EXISTS "${CMAKE_SOURCE_DIR}/data/runtime/${name}")
        configure_file("data/sample/${name}" "${CMAKE_SOURCE_DIR}/data/runtime/${name}" COPYONLY)
    endif()
endforeach()
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/data/runtime/test_routes_intermediate.csv")
    configure_file(data/test/test_routes_intermediate.csv "${CMAKE_SOURCE_DIR}/data/runtime/test_routes_intermediate.csv" COPYONLY)
endif()
